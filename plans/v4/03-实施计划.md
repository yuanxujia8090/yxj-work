    self.init()
    run(self.repo, "plan", "add", "u", "U", "--deps", "none",
        "--files", "README.md", "--verify", "true")
    run(self.repo, "next")
    run(self.repo, "verify", "u")

    run(self.repo, "plan", "add", "v", "V", "--deps", "none",

def test_inspect_requires_acceptance_source(self):
    self.init()
    run(self.repo, "plan", "add", "u", "U", "--deps", "none",
        "--files", "README.md", "--verify", "true")
    run(self.repo, "next")
    run(self.repo, "verify", "u")

    (self.repo / "req.md").write_text("需求：只读检查\n")

              "--source", "missing.md", "--units", "u")

    ok = run(self.repo, "acceptance", "set", "A1", "只读检查",

    self.assertFalse(out["done_met"])  # 缺最终集成单元
    self.assertEqual(out["acceptance_status"]["total"], 1)

    target = self.repo / "README.md"
    run(self.repo, "plan", "add", "u", "U", "--deps", "none",
        "--files", "README.md", "--verify", "true")

    stale = run(self.repo, "done", "u")

    self.assertNotEqual(run(self.repo, "resume").returncode, 0)
    self.assertEqual(run(self.repo, "resume", "--confirm",

## 5. 回滚

- 每个任务一个提交，`git revert <sha>` 即可单独回退。
- 回退前先 `xr pause` 所有活跃运行，保留运行目录与 `ACTIVE`。
- **不得**用备份覆盖 `ACTIVE`；若指针被改动，先读原/现两个运行与绑定，确认所有权后再恢复。
- 回退不删除 `plans/v4`、不重置用户未提交改动。

## 6. 评审门槛

实现完成后，独立只读审查必须覆盖三方面：

1. **证据与完成正确性**：`inspect` 是否可能误判完成；`stale` 是否会漏。
2. **授权与副作用**：是否存在未确认的自动重放、越权切换、绕过上限。
3. **兼容与复杂度**：legacy 是否被误判为 `managed`；新增文件是否真的必要。

审查报告与实现变更分开保存；父会话自审不能冒充独立审查。
